```python
from pydantic import BaseModel, Field
from typing import Optional, List
from datetime import datetime


class SentimentAnalyzeRequest(BaseModel):
    """
    Request model for sentiment analysis.

    WHY: We need a structured way to receive commit messages for analysis.
    HOW: Pydantic validates the input and provides automatic API documentation.
    """
    commit_message: str = Field(
        ...,  # Required field
        min_length=1,
        max_length=5000,
        description="The commit message to analyze for sentiment"
    )
    commit_id: Optional[str] = Field(
        None,
        description="Optional commit ID for tracking"
    )

    class Config:
        # Example for API documentation
        json_schema_extra = {
            "example": {
                "commit_message": "fix: resolved the annoying bug that was driving me crazy",
                "commit_id": "abc123"
            }
        }


class SentimentScore(BaseModel):
    """
    Detailed sentiment scores from VADER.

    WHY: VADER provides multiple sentiment dimensions, not just positive/negative.
    HOW: We expose all scores so consumers can make nuanced decisions.
    """
    positive: float = Field(
        ...,
        ge=0.0,  # Greater than or equal to 0
        le=1.0,  # Less than or equal to 1
        description="Positive sentiment score (0-1)"
    )
    negative: float = Field(
        ...,
        ge=0.0,
        le=1.0,
        description="Negative sentiment score (0-1)"
    )
    neutral: float = Field(
        ...,
        ge=0.0,
        le=1.0,
        description="Neutral sentiment score (0-1)"
    )
    compound: float = Field(
        ...,
        ge=-1.0,  # Compound score ranges from -1 to 1
        le=1.0,
        description="Compound sentiment score (-1 to 1, normalized)"
    )


class SentimentCategory(str):
    """
    Enum-like class for sentiment categories.

    WHY: Provides clear, consistent sentiment labels.
    HOW: String subclass allows easy serialization and comparison.
    """
    POSITIVE = "POSITIVE"
    NEGATIVE = "NEGATIVE"
    NEUTRAL = "NEUTRAL"
    VERY_POSITIVE = "VERY_POSITIVE"
    VERY_NEGATIVE = "VERY_NEGATIVE"


class SentimentAnalysisResult(BaseModel):
    """
    Complete sentiment analysis result.

    WHY: Provides both numerical scores and human-readable categories.
    HOW: Combines VADER scores with categorization logic.
    """
    commit_id: Optional[str] = Field(
        None,
        description="The commit ID that was analyzed"
    )
    sentiment: str = Field(
        ...,
        description="Overall sentiment category"
    )
    scores: SentimentScore = Field(
        ...,
        description="Detailed sentiment scores"
    )
    confidence: float = Field(
        ...,
        ge=0.0,
        le=1.0,
        description="Confidence in the sentiment classification (0-1)"
    )
    indicators: List[str] = Field(
        default_factory=list,
        description="Words or phrases that influenced the sentiment"
    )
    timestamp: datetime = Field(
        default_factory=datetime.utcnow,
        description="When the analysis was performed"
    )

    class Config:
        json_schema_extra = {
            "example": {
                "commit_id": "abc123",
                "sentiment": "NEGATIVE",
                "scores": {
                    "positive": 0.1,
                    "negative": 0.6,
                    "neutral": 0.3,
                    "compound": -0.5
                },
                "confidence": 0.75,
                "indicators": ["annoying", "crazy", "bug"],
                "timestamp": "2024-01-15T10:30:00Z"
            }
        }


class BatchSentimentRequest(BaseModel):
    """
    Request model for batch sentiment analysis.

    WHY: Analyzing multiple commits at once is more efficient.
    HOW: Accepts a list of messages and processes them in one request.
    """
    messages: List[str] = Field(
        ...,
        min_length=1,
        max_length=100,  # Limit batch size to prevent overload
        description="List of commit messages to analyze"
    )
    commit_ids: Optional[List[str]] = Field(
        None,
        description="Optional list of commit IDs (must match messages length)"
    )

    class Config:
        json_schema_extra = {
            "example": {
                "messages": [
                    "fix: resolved critical bug",
                    "feat: added amazing new feature",
                    "chore: updated dependencies"
                ],
                "commit_ids": ["abc123", "def456", "ghi789"]
            }
        }


class BatchSentimentResponse(BaseModel):
    """
    Response model for batch sentiment analysis.

    WHY: Returns results for all messages plus summary statistics.
    HOW: Aggregates individual results and calculates batch-level metrics.
    """
    results: List[SentimentAnalysisResult] = Field(
        ...,
        description="Individual sentiment analysis results"
    )
    summary: dict = Field(
        ...,
        description="Summary statistics for the batch"
    )
    processed_count: int = Field(
        ...,
        description="Number of messages successfully processed"
    )

    class Config:
        json_schema_extra = {
            "example": {
                "results": [
                    {
                        "commit_id": "abc123",
                        "sentiment": "NEGATIVE",
                        "scores": {"positive": 0.1, "negative": 0.6, "neutral": 0.3, "compound": -0.5},
                        "confidence": 0.75,
                        "indicators": ["critical", "bug"],
                        "timestamp": "2024-01-15T10:30:00Z"
                    }
                ],
                "summary": {
                    "positive_count": 1,
                    "negative_count": 1,
                    "neutral_count": 1,
                    "average_compound": 0.15
                },
                "processed_count": 3
            }
        }
```

**Line-by-Line Explanation:**

1. **Imports**: We import Pydantic for data validation, typing for type hints, and datetime for timestamps
2. **SentimentAnalyzeRequest**: Defines what data we need to analyze sentiment (commit message + optional ID)
3. **Field(..., min_length=1)**: The `...` means required, min_length ensures non-empty strings
4. **json_schema_extra**: Provides example data for Swagger UI documentation
5. **SentimentScore**: Breaks down VADER's multi-dimensional sentiment scores
6. **ge=0.0, le=1.0**: Validates that scores are between 0 and 1 (Pydantic constraint)
7. **compound score**: VADER's normalized score from -1 (very negative) to +1 (very positive)
8. **SentimentCategory**: String constants for sentiment labels (easier than enums for JSON)
9. **SentimentAnalysisResult**: Complete response with scores, category, and metadata
10. **indicators**: List of words that influenced the sentiment (helps explain the result)
11. **default_factory=datetime.utcnow**: Automatically sets timestamp when object is created
12. **BatchSentimentRequest**: Allows analyzing multiple messages in one API call
13. **max_length=100**: Prevents abuse by limiting batch size
14. **BatchSentimentResponse**: Returns individual results plus aggregate statistics

**Why This Structure:**
- **Separation of concerns**: Request, scores, and results are separate models
- **Validation**: Pydantic ensures data is correct before processing
- **Documentation**: Examples and descriptions auto-generate API docs
- **Type safety**: Type hints catch errors during development

#### Step 2.2: Create Sentiment Model

Create `app/models/sentiment.py`:

```python
from vaderSentiment.vaderSentiment import SentimentIntensityAnalyzer
from typing import List, Tuple, Dict
import re
from app.schemas.sentiment import (
    SentimentScore,
    SentimentAnalysisResult,
    SentimentCategory
)


class SentimentAnalyzer:
    """
    Sentiment analysis model using VADER.

    WHY: Analyzes emotional tone of commit messages to identify developer frustration,
         positive contributions, or neutral maintenance work.

    HOW: Uses VADER (Valence Aware Dictionary and sEntiment Reasoner), a lexicon and
         rule-based sentiment analysis tool specifically attuned to sentiments expressed
         in social media.
    """

    def __init__(self):
        """
        Initialize the VADER sentiment analyzer.

        WHY: VADER needs to be initialized once and reused for all analyses.
        HOW: Creates a SentimentIntensityAnalyzer instance that loads the lexicon.
        """
        self.analyzer = SentimentIntensityAnalyzer()

        # Custom sentiment indicators for commit messages
        # WHY: Commit messages have domain-specific words that indicate sentiment
        # HOW: We maintain lists of positive and negative indicators
        self.positive_indicators = [
            'fix', 'fixed', 'resolve', 'resolved', 'improve', 'improved',
            'optimize', 'optimized', 'enhance', 'enhanced', 'add', 'added',
            'implement', 'implemented', 'complete', 'completed', 'success',
            'working', 'clean', 'refactor', 'refactored', 'better'
        ]

        self.negative_indicators = [
            'bug', 'error', 'fail', 'failed', 'broken', 'issue', 'problem',
            'crash', 'critical', 'urgent', 'hack', 'workaround', 'temporary',
            'todo', 'fixme', 'xxx', 'annoying', 'frustrating', 'stupid',
            'damn', 'wtf', 'shit', 'crap', 'mess'
        ]

    def analyze(self, text: str, commit_id: str = None) -> SentimentAnalysisResult:
        """
        Analyze sentiment of a single text.

        WHY: Provides detailed sentiment analysis with scores and categorization.
        HOW: Uses VADER for scoring, then applies custom logic for categorization.

        Args:
            text: The commit message to analyze
            commit_id: Optional commit identifier

        Returns:
            SentimentAnalysisResult with scores, category, and indicators
        """
        # Get VADER scores
        # WHY: VADER returns a dictionary with pos, neg, neu, and compound scores
        # HOW: polarity_scores() analyzes the text and returns normalized scores
        scores = self.analyzer.polarity_scores(text)

        # Create SentimentScore object
        # WHY: We wrap VADER's dict in our Pydantic model for validation
        # HOW: Extract each score and create the model
        sentiment_scores = SentimentScore(
            positive=scores['pos'],
            negative=scores['neg'],
            neutral=scores['neu'],
            compound=scores['compound']
        )

        # Categorize sentiment based on compound score
        # WHY: Compound score is VADER's normalized overall sentiment (-1 to +1)
        # HOW: Use thresholds to categorize into 5 levels
        sentiment_category = self._categorize_sentiment(scores['compound'])

        # Calculate confidence
        # WHY: Higher absolute compound scores indicate more confident predictions
        # HOW: Use absolute value of compound score as confidence proxy
        confidence = self._calculate_confidence(scores)

        # Extract sentiment indicators
        # WHY: Helps explain why the sentiment was classified this way
        # HOW: Find words from our indicator lists in the text
        indicators = self._extract_indicators(text)

        return SentimentAnalysisResult(
            commit_id=commit_id,
            sentiment=sentiment_category,
            scores=sentiment_scores,
            confidence=confidence,
            indicators=indicators
        )

    def _categorize_sentiment(self, compound_score: float) -> str:
        """
        Categorize sentiment based on compound score.

        WHY: Provides human-readable sentiment labels instead of just numbers.
        HOW: Uses threshold-based classification on the compound score.

        Thresholds:
        - >= 0.5: VERY_POSITIVE (enthusiastic, excited)
        - >= 0.05: POSITIVE (satisfied, constructive)
        - > -0.05: NEUTRAL (factual, descriptive)
        - > -0.5: NEGATIVE (frustrated, concerned)
        - <= -0.5: VERY_NEGATIVE (angry, critical)

        Args:
            compound_score: VADER compound score (-1 to 1)

        Returns:
            Sentiment category string
        """
        if compound_score >= 0.5:
            return SentimentCategory.VERY_POSITIVE
        elif compound_score >= 0.05:
            return SentimentCategory.POSITIVE
        elif compound_score > -0.05:
            return SentimentCategory.NEUTRAL
        elif compound_score > -0.5:
            return SentimentCategory.NEGATIVE
        else:
            return SentimentCategory.VERY_NEGATIVE

    def _calculate_confidence(self, scores: Dict[str, float]) -> float:
        """
        Calculate confidence in the sentiment classification.

        WHY: Not all sentiment predictions are equally reliable.
        HOW: Use the absolute compound score and the dominance of one sentiment.

        High confidence indicators:
        - High absolute compound score (strong sentiment)
        - One sentiment dimension dominates (clear signal)

        Low confidence indicators:
        - Compound score near zero (mixed sentiment)
        - Multiple sentiment dimensions similar (ambiguous)

        Args:
            scores: VADER scores dictionary

        Returns:
            Confidence score (0-1)
        """
        # Base confidence from absolute compound score
        # WHY: Stronger sentiments are more confident
        # HOW: Absolute value maps -1/+1 to high confidence, 0 to low confidence
        base_confidence = abs(scores['compound'])

        # Adjust for sentiment clarity
        # WHY: If one sentiment dominates, we're more confident
        # HOW: Calculate the difference between max and second-max sentiment
        sentiment_values = [scores['pos'], scores['neg'], scores['neu']]
        sentiment_values.sort(reverse=True)

        # If top sentiment is much higher than second, increase confidence
        # WHY: Clear dominant sentiment = higher confidence
        # HOW: Difference of 0.3+ adds up to 0.2 confidence
        if len(sentiment_values) >= 2:
            clarity_bonus = min(0.2, (sentiment_values[0] - sentiment_values[1]) * 0.5)
            base_confidence = min(1.0, base_confidence + clarity_bonus)

        return round(base_confidence, 2)

    def _extract_indicators(self, text: str) -> List[str]:
        """
        Extract words that indicate sentiment.

        WHY: Provides explainability - shows which words influenced the sentiment.
        HOW: Searches for known positive/negative words in the text.

        Args:
            text: The commit message

        Returns:
            List of indicator words found in the text
        """
        # Convert to lowercase for matching
        # WHY: Case-insensitive matching (Bug = bug = BUG)
        # HOW: Simple .lower() transformation
        text_lower = text.lower()

        indicators = []

        # Find positive indicators
        # WHY: Shows what made the sentiment positive
        # HOW: Check if each positive word appears in the text
        for word in self.positive_indicators:
            if word in text_lower:
                indicators.append(word)

        # Find negative indicators
        # WHY: Shows what made the sentiment negative
        # HOW: Check if each negative word appears in the text
        for word in self.negative_indicators:
            if word in text_lower:
                indicators.append(word)

        # Remove duplicates and limit to top 10
        # WHY: Keep response size reasonable
        # HOW: Convert to set (removes duplicates) then back to list
        indicators = list(set(indicators))[:10]

        return indicators

    def analyze_batch(self, texts: List[str], commit_ids: List[str] = None) -> List[SentimentAnalysisResult]:
        """
        Analyze sentiment for multiple texts.

        WHY: More efficient than calling analyze() multiple times.
        HOW: Iterates through texts and analyzes each one.

        Args:
            texts: List of commit messages
            commit_ids: Optional list of commit IDs (must match texts length)

        Returns:
            List of SentimentAnalysisResult objects
        """
        # Validate commit_ids length if provided
        # WHY: Prevent mismatched data
        # HOW: Check lengths match or commit_ids is None
        if commit_ids and len(commit_ids) != len(texts):
            raise ValueError(
                f"commit_ids length ({len(commit_ids)}) must match texts length ({len(texts)})"
            )

        results = []

        # Analyze each text
        # WHY: Process all texts in the batch
        # HOW: Loop through texts with optional commit_ids
        for i, text in enumerate(texts):
            commit_id = commit_ids[i] if commit_ids else None
            result = self.analyze(text, commit_id)
            results.append(result)

        return results

    def get_batch_summary(self, results: List[SentimentAnalysisResult]) -> Dict:
        """
        Calculate summary statistics for a batch of results.

        WHY: Provides aggregate insights across multiple commits.
        HOW: Counts sentiment categories and calculates averages.

        Args:
            results: List of sentiment analysis results

        Returns:
            Dictionary with summary statistics
        """
        if not results:
            return {
                "positive_count": 0,
                "negative_count": 0,
                "neutral_count": 0,
                "very_positive_count": 0,
                "very_negative_count": 0,
                "average_compound": 0.0,
                "average_confidence": 0.0
            }

        # Count each sentiment category
        # WHY: Shows distribution of sentiments
        # HOW: Count occurrences of each category
        sentiment_counts = {
            "positive_count": 0,
            "negative_count": 0,
            "neutral_count": 0,
            "very_positive_count": 0,
            "very_negative_count": 0
        }

        total_compound = 0.0
        total_confidence = 0.0

        for result in results:
            # Count sentiment categories
            if result.sentiment == SentimentCategory.VERY_POSITIVE:
                sentiment_counts["very_positive_count"] += 1
            elif result.sentiment == SentimentCategory.POSITIVE:
                sentiment_counts["positive_count"] += 1
            elif result.sentiment == SentimentCategory.NEUTRAL:
                sentiment_counts["neutral_count"] += 1
            elif result.sentiment == SentimentCategory.NEGATIVE:
                sentiment_counts["negative_count"] += 1
            elif result.sentiment == SentimentCategory.VERY_NEGATIVE:
                sentiment_counts["very_negative_count"] += 1

            # Sum for averages
            total_compound += result.scores.compound
            total_confidence += result.confidence

        # Calculate averages
        # WHY: Provides overall sentiment trend
        # HOW: Divide totals by count
        count = len(results)
        sentiment_counts["average_compound"] = round(total_compound / count, 2)
        sentiment_counts["average_confidence"] = round(total_confidence / count, 2)

        return sentiment_counts


# Create singleton instance
# WHY: Reuse the same analyzer instance across requests (more efficient)
# HOW: Create one instance that can be imported elsewhere
sentiment_analyzer = SentimentAnalyzer()
```



**Line-by-Line Explanation:**

1. **Import VADER**: SentimentIntensityAnalyzer is the main VADER class
2. **SentimentAnalyzer class**: Wraps VADER with custom logic for commit messages
3. **__init__**: Initializes VADER once (loading lexicon is expensive)
4. **positive_indicators**: Words that typically indicate positive commits (fix, improve, etc.)
5. **negative_indicators**: Words that indicate problems or frustration
6. **analyze()**: Main method that analyzes a single commit message
7. **polarity_scores()**: VADER's core method that returns sentiment scores
8. **scores dict**: Contains 'pos', 'neg', 'neu' (0-1) and 'compound' (-1 to +1)
9. **_categorize_sentiment()**: Converts compound score to human-readable category
10. **Thresholds**: >=0.5 very positive, >=0.05 positive, >-0.05 neutral, etc.
11. **_calculate_confidence()**: Estimates how confident we are in the prediction
12. **abs(compound)**: Stronger sentiments (closer to -1 or +1) = higher confidence
13. **clarity_bonus**: If one sentiment dominates, we're more confident
14. **_extract_indicators()**: Finds words that explain the sentiment
15. **text_lower**: Case-insensitive matching for indicator words
16. **analyze_batch()**: Processes multiple messages efficiently
17. **get_batch_summary()**: Aggregates statistics across multiple results
18. **sentiment_analyzer**: Singleton instance for reuse

**Why This Implementation:**
- **VADER**: Fast, accurate for short text, no training needed
- **Custom indicators**: Domain-specific words for commit messages
- **Confidence scoring**: Helps consumers know when to trust the prediction
- **Explainability**: Indicators show which words influenced the result
- **Batch processing**: More efficient for analyzing many commits

```python
from fastapi import APIRouter, HTTPException, status
from typing import List
from app.schemas.sentiment import (
    SentimentAnalyzeRequest,
    SentimentAnalysisResult,
    BatchSentimentRequest,
    BatchSentimentResponse
)
from app.models.sentiment import sentiment_analyzer
import logging

# Set up logging
# WHY: Track API usage and errors for debugging
# HOW: Use Python's logging module
logger = logging.getLogger(__name__)

# Create router
# WHY: Organizes sentiment endpoints under /api/ml/sentiment
# HOW: FastAPI APIRouter with prefix and tags
router = APIRouter(
    prefix="/api/ml/sentiment",
    tags=["sentiment"],
    responses={
        500: {"description": "Internal server error"},
        400: {"description": "Bad request"}
    }
)


@router.post(
    "/analyze",
    response_model=SentimentAnalysisResult,
    status_code=status.HTTP_200_OK,
    summary="Analyze sentiment of a commit message",
    description="""
    Analyzes the sentiment of a single commit message using VADER sentiment analysis.

    Returns:
    - Sentiment category (VERY_POSITIVE, POSITIVE, NEUTRAL, NEGATIVE, VERY_NEGATIVE)
    - Detailed scores (positive, negative, neutral, compound)
    - Confidence score (0-1)
    - Indicator words that influenced the sentiment

    Use this endpoint to:
    - Identify frustrated developers (negative sentiment)
    - Track team morale over time
    - Flag potentially problematic commits
    - Understand emotional context of code changes
    """
)
```

## Sentiment Categories

- **VERY_POSITIVE**: Compound score >= 0.5 (enthusiastic, excited)
- **POSITIVE**: Compound score >= 0.05 (satisfied, constructive)
- **NEUTRAL**: Compound score > -0.05 (factual, descriptive)
- **NEGATIVE**: Compound score > -0.5 (frustrated, concerned)
- **VERY_NEGATIVE**: Compound score <= -0.5 (angry, critical)

## Use Cases

1. **Developer Morale Tracking**: Monitor sentiment trends over time
2. **Frustration Detection**: Identify when developers are struggling
3. **Code Quality Indicators**: Negative sentiment may indicate problematic areas
4. **Team Health Metrics**: Aggregate sentiment across teams

## Troubleshooting

### Issue: VADER not found
**Symptom:** `ModuleNotFoundError: No module named 'vaderSentiment'`

**Solution:**
```bash
pip install vaderSentiment
```

### Issue: All sentiments are NEUTRAL
**Symptom:** Every message returns NEUTRAL sentiment

**Solution:**
- Check that VADER is installed correctly
- Verify the message has actual content
- Try a message with strong sentiment words
- Check VADER lexicon loaded: `sentiment_analyzer.analyzer.lexicon`

### Issue: Low confidence scores
**Symptom:** All confidence scores are very low

**Solution:**
- This is normal for neutral messages
- Use messages with stronger sentiment words
- Check that indicator extraction is working
- Verify compound scores are reasonable

### Issue: Indicators not found
**Symptom:** `indicators` list is always empty

**Solution:**
- Check that indicator lists are defined in `__init__`
- Verify case-insensitive matching is working
- Add more domain-specific indicators
- Check that `_extract_indicators()` is being called

### Issue: Batch endpoint times out
**Symptom:** Batch requests take too long or timeout

**Solution:**
- Reduce batch size (max 100)
- Check for very long messages
- Verify VADER is initialized once (singleton)
- Consider adding async processing for large batches

### Issue: Tests failing
**Symptom:** Some tests fail intermittently

**Solution:**
- Check that test expectations match VADER behavior
- VADER scores can vary slightly with updates
- Use ranges instead of exact values in assertions
- Verify test data has clear sentiment
